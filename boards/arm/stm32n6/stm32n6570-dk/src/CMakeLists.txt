set(SRCS stm32_boot.c stm32_bringup.c)

if(CONFIG_ARCH_LEDS)
  list(APPEND SRCS stm32_autoleds.c)
else()
  list(APPEND SRCS stm32_userleds.c)
endif()

if(CONFIG_ARCH_BUTTONS)
  list(APPEND SRCS stm32_buttons.c)
endif()

if(CONFIG_PWM)
  list(APPEND SRCS stm32_pwm.c)
endif()

if(CONFIG_STM32_ADC1)
  list(APPEND SRCS stm32_adc.c)
endif()

if(CONFIG_STM32_SPI)
  list(APPEND SRCS stm32_spi.c)
endif()

if(CONFIG_STM32_LTDC)
  list(APPEND SRCS stm32_lcd.c)
endif()

if(CONFIG_STM32_SDMMC2)
  list(APPEND SRCS stm32_sdmmc.c)
endif()

if(CONFIG_STM32_OTGHS AND CONFIG_USBHOST)
  list(APPEND SRCS stm32_usbhost.c)
endif()

if(CONFIG_STM32N6570_DK_PSRAM)
  list(APPEND SRCS stm32_psram.c)
endif()

if(CONFIG_STM32N6570_DK_NOR)
  list(APPEND SRCS stm32_mx66uw1g45g.c)
endif()

if(CONFIG_STM32_DCMIPP OR CONFIG_SENSORS_LSM6DS3TRC)
  list(APPEND SRCS stm32_camera.c)
endif()

if(CONFIG_AUDIO_WM8904)
  list(APPEND SRCS stm32_wm8904.c)
endif()

if(CONFIG_STM32_MDF1)
  list(APPEND SRCS stm32_mdf.c)
endif()

target_sources(board PRIVATE ${SRCS})

set_property(GLOBAL PROPERTY LD_SCRIPT "${NUTTX_BOARD_DIR}/scripts/flash.ld")
